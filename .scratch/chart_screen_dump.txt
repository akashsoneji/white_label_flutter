401: // ── stacked painter: candle (top) + MACD (bottom) ──────────────────────────
402: class _StackedChartPainter extends CustomPainter {
403:   _StackedChartPainter({
404:     required this.candles,
405:     required this.ma5,
406:     required this.ma10,
407:     required this.ma30,
408:     required this.macdHist,
409:     required this.macdDif,
410:     required this.macdDea,
411:     required this.priceDigits,
412:     required this.crosshairX,
413:     required this.crosshairY,
414:     required this.showCrosshair,
415:   });
416: 
417:   final List<IQuoteCandle> candles;
418:   final List<double?> ma5;
419:   final List<double?> ma10;
420:   final List<double?> ma30;
421:   final List<double?> macdHist;
422:   final List<double?> macdDif;
423:   final List<double?> macdDea;
424:   final int priceDigits;
425:   final double? crosshairX;
426:   final double? crosshairY;
427:   final bool showCrosshair;
428: 
429:   @override
430:   void paint(Canvas canvas, Size size) {
431:     if (candles.isEmpty) return;
432:     final w = size.width;
433:     final macdH = size.height * 0.28;
434:     final chartH = size.height - macdH;
435:     final chartCY = (crosshairY == null || crosshairY! > chartH) ? null : crosshairY;
436: 
437:     canvas.save();
438:     canvas.clipRect(Rect.fromLTWH(0, 0, w, chartH));
439:     canvas.drawPaint(Paint()..color = const Color(0xFF101722));
440: 
441:     final candlePainter = CandleChartPainter(
442:       candles: candles,
443:       ma5: ma5,
444:       ma10: ma10,
445:       ma30: ma30,
446:       crosshairX: crosshairX,
447:       crosshairY: chartCY,
448:       showCrosshair: showCrosshair,
449:       priceDigits: priceDigits,
450:     );
451:     candlePainter.paint(canvas, Size(w, chartH));
452:     canvas.restore();
453: 
454:     final divPaint = Paint()
455:       ..color = const Color(0x22FFFFFF)
456:       ..strokeWidth = 1;
457:     canvas.drawLine(Offset(0, chartH - 0.5), Offset(w, chartH - 0.5), divPaint);
458: 
459:     canvas.save();
460:     canvas.translate(0, chartH);
461:     final macdPainter = MacdPainter(
462:       hist: macdHist,
463:       dif: macdDif,
464:       dea: macdDea,
465:     );
466:     macdPainter.paint(canvas, Size(w, macdH));
467:     canvas.restore();
468:   }
469: 
470:   @override
471:   bool shouldRepaint(_StackedChartPainter old) => true;

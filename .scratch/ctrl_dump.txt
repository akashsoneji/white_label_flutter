 56: // ── controller ───────────────────────────────────────────────────────────────
 57: class IQuoteChartController extends GetxController {
 58:   IQuoteChartController(this._symbol);
 59:   final String _symbol;
 60:   String get symbol => _symbol;
 61: 
 62:   final RxList<IQuoteCandle> candles = <IQuoteCandle>[].obs;
 63:   final RxList<double?> ma5 = <double?>[].obs;
 64:   final RxList<double?> ma10 = <double?>[].obs;
 65:   final RxList<double?> ma30 = <double?>[].obs;
 66:   final RxList<double?> macd = <double?>[].obs;
 67:   final RxList<double?> macdSignal = <double?>[].obs;
 68:   final RxList<double?> macdHist = <double?>[].obs;
 69: 
 70:   final RxnDouble lastBid = RxnDouble();
 71:   final RxnDouble lastAsk = RxnDouble();
 72:   RxnDouble openPrice = RxnDouble();
 73:   final RxInt digits = 0.obs;
 74: 
 75:   final RxString period = kDefaultPeriod.obs;
 76:   final RxBool loading = true.obs;
 77:   final RxnString errorMsg = RxnString();
 78: 
 79:   StreamSubscription<dynamic>? _priceSub;
 80:   final Map<int, double> _ticks = <int, double>{};
 81:   int? _currentMinute;
 82: 
 83:   // ── lifecycle ──────────────────────────────────────────────────────────────
 84:   @override
 85:   void onInit() {
 86:     super.onInit();
 87:     _fetchHistory().then((_) => _bindPriceStream());
 88:   }
 89: 
 90:   @override
 91:   void onClose() {
 92:     _priceSub?.cancel();
 93:     super.onClose();
 94:   }

 96:   // ── period ─────────────────────────────────────────────────────────────────
 97:   void changePeriod(String p) {
 98:     if (!kPeriods.contains(p) || period.value == p) return;
 99:     period.value = p;
100:     _currentMinute = null;
101:     _ticks.clear();
102:     _fetchHistory();
103:   }
104: 
105:   // ── price stream ───────────────────────────────────────────────────────────
106:   void _bindPriceStream() {
107:     final ctrl = Get.find<ChartSocketController>();
108:     ctrl.initializeSocketForSymbol(_symbol);
109:     final bid = ctrl.getBidPrice();
110:     if (bid != null) openPrice.value = bid;
111: 
112:     _priceSub = ctrl.priceStream.listen((Map<String, dynamic> raw) {
113:       final sym = raw['symbol'];
114:       if (sym is! String || sym != _symbol) return;
115:       final bidRaw = raw['bid'];
116:       final askRaw = raw['ask'];
117:       final digitsRaw = raw['digits'];
118:       final bid = (bidRaw is num) ? bidRaw.toDouble() : null;
119:       final ask = (askRaw is num) ? askRaw.toDouble() : null;
120:       final dStr = (digitsRaw is String) ? digitsRaw : '5';
121:       digits.value = int.tryParse(dStr) ?? 5;
122:       if (bid != null) lastBid.value = bid;
123:       if (ask != null) lastAsk.value = ask;
124:       _aggregateTick(bid ?? ask ?? 0.0);
125:     });
126:   }
127: 
128:   void _aggregateTick(double price) {
129:     final secs = periodToSeconds(period.value);
130:     final minute = DateTime.now().millisecondsSinceEpoch ~/ 1000 ~/ secs;
131: 
132:     if (_currentMinute != minute) {
133:       _currentMinute = minute;
134:       _ticks[minute] = price;
135:       _addMinuteCandle(minute, price);
136:     } else {
137:       _ticks[minute] = price;
138:       _updateCurrentCandle(price);
139:     }
140:   }
141: 
142:   void _addMinuteCandle(int minute, double price) {
143:     if (candles.isEmpty) return;
144:     final secs = periodToSeconds(period.value);
145:     final start = DateTime.fromMillisecondsSinceEpoch(minute * secs * 1000, isUtc: true);
146:     candles.add(IQuoteCandle(time: start, open: price, high: price, low: price, close: price));
147:     _recomputeAll();
148:   }
149: 
150:   void _updateCurrentCandle(double price) {
151:     if (candles.isEmpty) return;
152:     final last = candles.last;
153:     last.close = price;
154:     if (price > last.high) last.high = price;
155:     if (price < last.low) last.low = price;
156:     _recomputeAll();
157:   }

159:   // ── history ────────────────────────────────────────────────────────────────
160:   Future<void> _fetchHistory() async {
161:     loading.value = true;
162:     errorMsg.value = null;
163:     candles.clear();
164:     ma5.clear(); ma10.clear(); ma30.clear();
165:     macd.clear(); macdSignal.clear(); macdHist.clear();
166: 
167:     try {
168:       final raw = await _fetchFromApi();
169:       if (raw.isNotEmpty) { _loadCandles(raw); return; }
170:     } on Exception catch (_) {}
171: 
172:     _generateSyntheticCandles();
173:   }
174: 
175:   Future<List<IQuoteCandle>> _fetchFromApi() async {
176:     final end = DateTime.now().millisecondsSinceEpoch ~/ 1000;
177:     const count = 200;
178:     final secs = periodToSeconds(period.value);
179:     final from = end - count * secs;
180: 
181:     final res = await ApiService.get(
182:       endpoint: '/api/market/history',
183:       queryParameters: {
184:         'symbol': _symbol,
185:         'from': '$from',
186:         'to': '$end',
187:         'count': '$count',
188:         'resolution': period.value,
189:       },
190:     );
191: 
192:     if (!res.success || res.data == null) return const [];
193: 
194:     final Map<String, dynamic> body = res.data is Map
195:         ? res.data as Map<String, dynamic>
196:         : <String, dynamic>{};
197: 
198:     final times = (body['t'] as List?) ?? const [];
199:     final opens = (body['o'] as List?) ?? const [];
200:     final highs = (body['h'] as List?) ?? const [];
201:     final lows  = (body['l'] as List?) ?? const [];
202:     final closes = (body['c'] as List?) ?? const [];
203: 
204:     final len = [times.length, opens.length, highs.length, lows.length, closes.length]
205:         .fold<int>(times.length, (a, b) => a < b ? a : b);
206: 
207:     final out = <IQuoteCandle>[];
208:     for (var i = 0; i < len; i++) {
209:       final epoch = _toEpoch(times[i]);
210:       if (epoch == null) continue;
211:       out.add(IQuoteCandle(
212:         time: DateTime.fromMillisecondsSinceEpoch(epoch * 1000, isUtc: true),
213:         open: _toD(opens[i]),
214:         high: _toD(highs[i]),
215:         low: _toD(lows[i]),
216:         close: _toD(closes[i]),
217:       ));
218:     }
219:     return out;
220:   }

222:   void _loadCandles(List<IQuoteCandle> raw) {
223:     if (raw.isEmpty) { _generateSyntheticCandles(); return; }
224:     candles.assignAll(raw);
225:     if (openPrice.value == null) openPrice.value = raw.first.open;
226:     _recomputeAll();
227:     loading.value = false;
228:   }
229: 
230:   void _generateSyntheticCandles() {
231:     final now = DateTime.now().toUtc();
232:     final secs = periodToSeconds(period.value);
233:     double price = 2000 + Random().nextDouble() * 200;
234:     final out = <IQuoteCandle>[];
235:     for (var i = 199; i >= 0; i--) {
236:       final t = now.subtract(Duration(seconds: i * secs));
237:       final vol = 3 + (i % 5);
238:       final up = Random().nextBool();
239:       final c = price + (up ? 1 : -1) * Random().nextDouble() * vol;
240:       final h = max(price, c) + Random().nextDouble() * vol * 0.5;
241:       final l = min(price, c) - Random().nextDouble() * vol * 0.5;
242:       out.add(IQuoteCandle(time: t, open: price, high: h, low: l, close: c));
243:       price = c;
244:     }
245:     candles.assignAll(out);
246:     if (openPrice.value == null) openPrice.value = out.first.open;
247:     _recomputeAll();
248:     loading.value = false;
249:   }

251:   // ── moving averages ─────────────────────────────────────────────────────────
252:   static double? _sma(List<double> src, int period, int end) {
253:     if (end < period - 1) return null;
254:     double s = 0;
255:     for (var i = end - period + 1; i <= end; i++) s += src[i];
256:     return s / period;
257:   }
258: 
259:   void _recomputeAll() {
260:     final n = candles.length;
261:     final closes = candles.map((c) => c.close).toList();
262:     final a5 = <double?>[], a10 = <double?>[], a30 = <double?>[];
263:     for (var i = 0; i < n; i++) {
264:       a5.add(_sma(closes, kMaFast, i));
265:       a10.add(_sma(closes, kMaSlow, i));
266:       a30.add(_sma(closes, kMaLong, i));
267:     }
268:     ma5.assignAll(a5); ma10.assignAll(a10); ma30.assignAll(a30);
269:     final (m, s, h) = _computeMacd(closes);
270:     macd.assignAll(m); macdSignal.assignAll(s); macdHist.assignAll(h);
271:   }
272: 
273:   static (List<double?>, List<double?>, List<double?>) _computeMacd(List<double> closes) {
274:     final n = closes.length;
275:     if (n == 0) return (<double?>[], <double?>[], <double?>[]);
276: 
277:     double ema(List<double> src, int p, int idx) {
278:       if (idx == 0) return src[0];
279:       final k = 2.0 / (p + 1);
280:       double v = src[0];
281:       for (var i = 1; i <= idx; i++) v = src[i] * k + v * (1 - k);
282:       return v;
283:     }
284: 
285:     final fastEma = <double>[], slowEma = <double>[];
286:     for (var i = 0; i < n; i++) {
287:       fastEma.add(ema(closes, kMacdFast, i));
288:       slowEma.add(ema(closes, kMacdSlow, i));
289:     }
290:     final macdLine = <double?>[];
291:     for (var i = 0; i < n; i++) macdLine.add(fastEma[i] - slowEma[i]);
292: 
293:     final signalLine = <double?>[];
294:     double? prevSig;
295:     for (final v in macdLine) {
296:       if (v == null) { signalLine.add(null); continue; }
297:       final k = 2.0 / (kMacdSignal + 1);
298:       final sig = (prevSig == null ? v : v * k + prevSig * (1 - k));
299:       prevSig = sig;
300:       signalLine.add(sig);
301:     }
302:     final hist = <double?>[];
303:     for (var i = 0; i < n; i++) {
304:       final m = macdLine[i], s = signalLine[i];
305:       hist.add((m == null || s == null) ? null : m - s);
306:     }
307:     return (macdLine, signalLine, hist);
308:   }

310:   // ── helpers ────────────────────────────────────────────────────────────────
311:   static int periodToSeconds(String p) {
312:     final v = int.tryParse(p);
313:     if (v != null && v > 0) return v * 60;
314:     return 60;
315:   }
316:   static int? _toEpoch(dynamic v) {
317:     if (v is int) return v;
318:     if (v is num) return v.toInt();
319:     if (v is String) return int.tryParse(v);
320:     return null;
321:   }
322:   static double _toD(dynamic v) {
323:     if (v is num) return v.toDouble();
324:     if (v is String) return double.tryParse(v) ?? 0.0;
325:     return 0.0;
326:   }

336:     final p = currentPrice;

342:   double? get priceChange {
343:     if (openPrice.value == null || currentPrice == null) return null;

348:     final open = openPrice.value;
349:     if (open == null || open == 0) return null;
